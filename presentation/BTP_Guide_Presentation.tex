\begin{frame}{1. Problem: Speculative Prefetch Interference in Shared LLCs}
  \vspace{-0.12cm}
  \begin{columns}[t, onlytextwidth]
    \begin{column}{0.485\textwidth}
      \begin{tcolorbox}[tealpanel]
        {\footnotesize\textbf{\color{Obsidian}Regular / Stride Co-Runners (\texttt{gcc}, \texttt{perlbench})}}\par\vspace{0.08cm}
        \begin{itemize}\setlength{\itemsep}{0.22em}\scriptsize
          \item \textbf{Memory Locality:} Sequential array strides and compact L1/L2 working sets (\texttt{IPC} $\approx 1.83\text{--}2.07$).
          \item \textbf{Prefetch Utility:} Hardware L2 stride/AMPM prefetchers hide 200+ cycle off-chip DRAM miss stalls.
          \item \textbf{Sensitivity:} Vulnerable when shared 8\,MB LLC capacity is displaced by noisy neighbor cores.
        \end{itemize}
      \end{tcolorbox}
    \end{column}%
    \hfill%
    \begin{column}{0.485\textwidth}
      \begin{tcolorbox}[archpanel]
        {\footnotesize\textbf{\color{Obsidian}Irregular Co-Runners (\texttt{mcf}, \texttt{omnetpp})}}\par\vspace{0.08cm}
        \begin{itemize}\setlength{\itemsep}{0.22em}\scriptsize
          \item \textbf{Memory Locality:} Linked pointer chasing and irregular heap traversals (\texttt{IPC} $\approx 0.08\text{--}0.34$).
          \item \textbf{Bandwidth Pressure:} L2 prefetcher achieves only $27\text{--}32\%$ accuracy; $>68\%$ of speculative fills miss reuse.
          \item \textbf{Capacity Thrashing:} Unneeded prefetch fills occupy shared LLC ways before aging out.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.22cm}

  \begin{tcolorbox}[summarystrip]
    {\footnotesize\textbf{\color{NavyPrimary}Root Cause — Prefetch-Oblivious Replacement:}}\par\vspace{0.06cm}
    \scriptsize
    Standard policies (\texttt{LRU}, \texttt{SRRIP}) ignore the \texttt{access\_type} signal during LLC cache fills. Under \texttt{LRU}, speculative \texttt{mcf} prefetches enter at MRU and displace active working-set blocks of co-running threads (\texttt{omnetpp}, \texttt{gcc}). Even \texttt{SRRIP} treats speculative fills identically to demand loads ($RRPV = 2$), degrading \texttt{omnetpp} IPC in heterogeneous mixes.
  \end{tcolorbox}
\end{frame}

\begin{frame}{2. Literature Review \& Real Hardware Storage Accounting}
  \vspace{-0.15cm}
  \begin{columns}[t, onlytextwidth]

      \begin{tcolorbox}[cleanpanel]
        {\footnotesize\textbf{\color{Obsidian}LRU / SRRIP (ISCA'10)}}\par\vspace{0.05cm}
        \scriptsize
        Inserts all fills uniformly (MRU or $RRPV=2$). Zero extra SRAM, but vulnerable to speculative bursts.
      \end{tcolorbox}
    \end{column}%
    \hfill%
    \begin{column}{0.32\textwidth}
      \begin{tcolorbox}[cleanpanel]
        {\footnotesize\textbf{\color{Obsidian}DRRIP / PACMan (MICRO'11)}}\par\vspace{0.05cm}
        \scriptsize
        \texttt{DRRIP} uses set dueling (~16\,B \texttt{PSEL}). \texttt{PACMan-HM} demotes PF fills ($R=3$) and suppresses PF-hit promotion.
      \end{tcolorbox}
    \end{column}%
    \hfill%
    \begin{column}{0.32\textwidth}
      \begin{tcolorbox}[cleanpanel]
        {\footnotesize\textbf{\color{Obsidian}SHiP (ISCA'11)}}\par\vspace{0.05cm}
        \scriptsize
        Learns dead-block PC signatures via 8,192$\times$2b \texttt{SHCT} (2\,KB SRAM + sampler sets). High accuracy at area cost.
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.18cm}

  \centering
  \resizebox{\linewidth}{!}{%
  \begin{tabular}{@{}l c c c l@{}}

        \begin{itemize}\setlength{\itemsep}{0.18em}\footnotesize
          \item \textbf{Versus PACMan-HM / Dyn (MICRO'11):}\\
            \texttt{PACMan-HM} blocks prefetch-hit promotion (\texttt{type != PREFETCH}) and \texttt{PACMan-Dyn} uses 32-set Set Dueling (\texttt{PSEL}). \texttt{PA-SRRIP} keeps uniform hit promotion; \texttt{PA-Adaptive} tracks reuse via one global \texttt{+3/-1} counter.
          \item \textbf{Explicit Per-Line Bit Cost (\texttt{is\_prefetched}):}\\
            \texttt{PA-Adaptive} reads \texttt{is\_prefetched[index]} on evict/hit. Standalone storage for 131,072 lines is \textbf{16\,KB} ($8\times$ \texttt{SHiP}), so sub-\texttt{SHiP} overhead (\textbf{0.75\,B}) assumes reusing an existing PMU per-line prefetch bit.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}
