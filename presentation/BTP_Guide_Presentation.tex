\begin{frame}{1. Motivation \& The Prefetch Pollution Dilemma}
  \begin{columns}[c]
    \begin{column}{0.48\linewidth}
      \begin{block}{The Hardware Prefetching Dilemma}
        \begin{itemize}
          \item Hardware prefetchers (IP-Stride, AMPM, SPP) hide memory latency for regular spatial streams (\texttt{gcc}, \texttt{lbm}).
          \item For irregular / pointer-chasing patterns (\texttt{429.mcf}), prefetch accuracy drops to $<1\%$.
          \item Inaccurate prefetch blocks flood the shared Last-Level Cache (LLC), causing severe cache pollution.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\linewidth}
      \begin{block}{Multicore Amplification \& Policy Blindness}
        \begin{itemize}
          \item Standard replacement policies (LRU, SRRIP) are \textbf{prefetch-oblivious}.
          \item Speculative prefetches receive high priority ($RRPV=2$ / MRU), evicting active demand data of co-running cores.
          \item In shared 4-core LLCs, pointer noise thrashes cross-core capacity, causing up to \textbf{50\%+ IPC collapse}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Slide 3: Literature Review & Critical Research Gaps

\begin{frame}{2. Literature Review \& Identified Research Gaps}
  \begin{columns}[t]
    \begin{column}{0.48\linewidth}
      \begin{block}{State-of-the-Art Approaches}
        \begin{itemize}
          \item \textbf{Drishti (MICRO 2025):} Slicing-aware framework using per-core GPHT tables and sampled cache selection. Framework is replacement-agnostic and relies on underlying policies for prefetch handling.
          \item \textbf{PACMan (IEEE CAL 2023):} Prefetch-aware cache management via accuracy tracking; requires additional sampling counters.
          \item \textbf{DRRIP (ISCA 2010):} Dynamic RRIP using set-dueling PSEL counters.
          \item \textbf{SHiP (ISCA 2011):} Signature-based hit predictor requiring an \textbf{8,192-entry SHCT table} (2 KB SRAM).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\linewidth}
      \begin{block}{Critical Research Gaps}
        \begin{enumerate}
          \item \textbf{Lack of Zero-Overhead Solutions:} Existing schemes introduce dedicated SRAM tables or complex set dueling.
          \item \textbf{Monolithic vs. Sliced Gap:} Prefetch-replacement interactions in multicore shared LLCs require lightweight, self-adaptive mechanisms.
          \item \textbf{Core Goal:} Achieve prefetch resilience with \textbf{ZERO hardware storage overhead} (0 extra metadata bits).
        \end{enumerate}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Slide 4: Proposed Policy Architecture
% ------------------------------------------------------------------------------
\begin{frame}{3. Proposed Architecture: Zero-Overhead \& Adaptive Demotion}
  \begin{columns}[c]
    \begin{column}{0.50\linewidth}
      \begin{block}{Key Architectural Mechanisms}
        \begin{enumerate}
          \item \textbf{Demoted Initial Insertion:}
            \begin{itemize}
              \item Demand fills $\rightarrow$ Inserted at MRU / $RRPV=2$.
              \item Prefetch fills $\rightarrow$ Demoted to near-LRU / $RRPV=3$.
              \item Useless prefetches sit in victim position and exit quietly.
            \end{itemize}
          \item \textbf{Self-Adaptive Promotion on Hit:}
            \begin{itemize}
              \item On cache hit, prefetched line promoted to MRU / $RRPV=0$.
              \item Line is reclassified as verified demand data.
            \end{itemize}
          \item \textbf{Novel PA-Adaptive Extension (0.75 Bytes):}
            \begin{itemize}
              \item 6-bit saturating counter dynamically adapts insertion between $RRPV=2$ (streaming) and $RRPV=3$ (noise filtering).
            \end{itemize}
        \end{enumerate}
      \end{block}
    \end{column}

    \begin{column}{0.46\linewidth}
