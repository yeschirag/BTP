\begin{frame}{1. Motivation \& Multicore Cache Contention}
  \begin{columns}[t]
    % Column 1
    \begin{column}{0.49\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.4mm, bottom=1.4mm]
        {\small\textbf{\color{NavyPrimary}Hardware Prefetching Paradox}}\par\vspace{0.06cm}
        \begin{itemize}\setlength{\itemsep}{0.2em}\footnotesize
          \item \textbf{Regular Streams (\texttt{gcc}, \texttt{lbm}):}\\
            Spatial stride prefetchers achieve $>50\%$ accuracy, effectively hiding memory latency.
          \item \textbf{Irregular Streams (\texttt{429.mcf}):}\\
            Pointer chasing causes accuracy to drop below \textbf{$<1\%$}.
          \item \textbf{Aggressive Speculation:}\\
            Hardware prefetchers issue speculative requests without feedback on LLC utility.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    % Column 2
    \begin{column}{0.49\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.4mm, bottom=1.4mm]
        {\small\textbf{\color{NavyPrimary}Multicore Contention \& Blindness}}\par\vspace{0.06cm}
        \begin{itemize}\setlength{\itemsep}{0.2em}\footnotesize
          \item \textbf{Prefetch-Oblivious Policies:}\\
            Standard LRU and SRRIP insert prefetched lines at high priority ($RRPV=2$ / MRU).
          \item \textbf{Cross-Core Interference:}\\
            Inaccurate prefetches evict valid demand blocks of co-running compute cores.
          \item \textbf{Severe Throughput Loss:}\\
            Unchecked prefetch thrashing causes up to \textbf{50\%+ IPC collapse} in shared multicore LLCs.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[boxrule=0.8pt, arc=1.5mm, colback=HighlightBg, colframe=HighlightBorder, top=1.4mm, bottom=1.4mm]
    \centering\footnotesize
    \textbf{Research Goal:} \textit{Design an LLC policy that neutralizes prefetch pollution with \textbf{ZERO hardware storage overhead} (0 extra metadata bits).}
  \end{tcolorbox}
\end{frame}

\begin{frame}{2. Literature Review \& Identified Research Gaps}
  \begin{columns}[t]
    % Column 1: SOTA
    \begin{column}{0.50\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.4mm, bottom=1.4mm]
        {\small\textbf{\color{NavyPrimary}State-of-the-Art Approaches}}\par\vspace{0.06cm}
        \begin{itemize}\setlength{\itemsep}{0.2em}\footnotesize
          \item \textbf{PACMan (IEEE CAL 2023):}\\
            Prefetch-aware cache management via dual queues; requires dedicated accuracy counters and sampling sets.
          \item \textbf{DRRIP (ISCA 2010):}\\
            Dynamic RRIP using set-dueling PSEL counters; oblivious to prefetch vs.\ demand stream origin.
          \item \textbf{SHiP (ISCA 2011):}\\
            Signature-based hit predictor; requires an \textbf{8,192-entry SHCT table (2,048 Bytes SRAM)}.
          \item \textbf{Hawkeye (ISCA 2016):}\\
            Learns optimal cache replacement but requires extensive hardware history tables.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    % Column 2: Gaps
    \begin{column}{0.48\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.4mm, bottom=1.4mm]
        {\small\textbf{\color{NavyPrimary}Identified Research Gaps}}\par\vspace{0.06cm}
        \begin{enumerate}\setlength{\itemsep}{0.25em}\footnotesize
          \item \textbf{High Storage Overhead:}\\
            Existing predictive policies require dedicated SRAM tables (2KB in SHiP) or complex sampling logic.
          \item \textbf{Lack of Zero-Overhead Resilience:}\\
            Standard high-performance caches cannot afford extra SRAM tables solely for prefetch tracking.
          \item \textbf{Our Proposed Solution:}\\
            A \textbf{Zero-Overhead} policy (\texttt{PA-SRRIP}) and a \textbf{0.75-Byte Self-Tuning} controller (\texttt{PA-Adaptive}) that eliminates prefetch pollution right at insertion.
        \end{enumerate}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{3. Proposed Architecture: PA-SRRIP \& PA-Adaptive}
  \begin{columns}[c]
    % Left Column: Architecture
    \begin{column}{0.52\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.4mm, bottom=1.4mm]
        {\small\textbf{\color{NavyPrimary}Dual-Mechanism Architectural Design}}\par\vspace{0.06cm}
        {\footnotesize \textbf{1. Demoted Initial Insertion (0 Extra Bits):}}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item Demand fills $\rightarrow$ Inserted at normal priority ($RRPV=2$).
          \item Prefetch fills $\rightarrow$ Demoted to victim status ($RRPV=3$).
          \item Inaccurate blocks are evicted first without polluting LLC.
        \end{itemize}
        \vspace{0.08cm}
        {\footnotesize \textbf{2. Self-Adaptive Promotion on Demand Hit:}}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item On first demand hit, promoted to MRU ($RRPV=0$).
          \item Line is reclassified as verified demand data $\rightarrow$ shielded.
        \end{itemize}
        \vspace{0.08cm}
        {\footnotesize \textbf{3. Novel PA-Adaptive Extension (0.75 Bytes):}}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item 6-bit saturating counter ($0..63$) monitors LLC utility.
          \item Dynamically inserts at $RRPV=2$ when useful, $RRPV=3$ when noisy.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
