\begin{frame}{1. Motivation \& Multicore Cache Contention}
  \begin{columns}[t]
    % Column 1
    \begin{column}{0.49\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.4mm, bottom=1.4mm]
        {\small\textbf{\color{NavyPrimary}Hardware Prefetching Paradox}}\par\vspace{0.06cm}
        \begin{itemize}\setlength{\itemsep}{0.2em}\footnotesize
          \item \textbf{Regular Streams (\texttt{gcc}, \texttt{lbm}):}\\
            Spatial stride prefetchers achieve $>50\%$ accuracy, effectively hiding the 240-cycle DRAM latency.
          \item \textbf{Irregular Streams (\texttt{429.mcf}):}\\
            Unpredictable pointer chasing causes accuracy to collapse below \textbf{$<1\%$}.
          \item \textbf{Aggressive Speculation:}\\
            Core-level prefetchers blindly issue speculative requests into the shared LLC without utility feedback.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    % Column 2
    \begin{column}{0.49\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.4mm, bottom=1.4mm]
        {\small\textbf{\color{NavyPrimary}Multicore Contention \& Blindness}}\par\vspace{0.06cm}
        \begin{itemize}\setlength{\itemsep}{0.2em}\footnotesize
          \item \textbf{Prefetch-Oblivious Policies:}\\
            Standard LRU and SRRIP insert speculative prefetches with the same high priority as real demand loads.
          \item \textbf{Cross-Core Interference:}\\
            Inaccurate prefetches from memory-stalled cores evict valid demand blocks of co-running compute cores.
          \item \textbf{Throughput Degradation:}\\
            Unchecked prefetch noise causes severe IPC collapse across shared multicore LLCs.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{tcolorbox}[boxrule=0.8pt, arc=1.5mm, colback=HighlightBg, colframe=HighlightBorder, top=1.4mm, bottom=1.4mm]
    \centering\footnotesize
    \textbf{Research Question:} \textit{Can we design an LLC policy that neutralizes prefetch pollution with \textbf{ZERO hardware storage overhead} (0 extra metadata bits)?}
  \end{tcolorbox}
\end{frame}

\begin{frame}{2. Literature Review \& Identified Research Gaps}
  \begin{columns}[t]
    % Column 1: SOTA
    \begin{column}{0.50\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.4mm, bottom=1.4mm]
        {\small\textbf{\color{NavyPrimary}State-of-the-Art Approaches}}\par\vspace{0.06cm}
        \begin{itemize}\setlength{\itemsep}{0.2em}\footnotesize
          \item \textbf{PACMan (IEEE CAL 2023):}\\
            Dual-queue prefetch management; requires dedicated tracking counters and dynamic queue migration logic.
          \item \textbf{DRRIP (ISCA 2010):}\\
            Dynamic RRIP via set-dueling PSEL counters; remains prefetch-oblivious (treats demand and prefetch identically).
          \item \textbf{SHiP (ISCA 2011):}\\
            Signature-based hit predictor; requires an \textbf{8,192-entry SHCT table (2,048 Bytes SRAM)}.
          \item \textbf{Hawkeye (ISCA 2016):}\\
            Learns Belady's MIN replacement; requires extensive history tables and OPTgen hardware engines.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    % Column 2: Gaps
    \begin{column}{0.48\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.4mm, bottom=1.4mm]
        {\small\textbf{\color{NavyPrimary}Identified Research Gaps}}\par\vspace{0.06cm}
        \begin{enumerate}\setlength{\itemsep}{0.25em}\footnotesize
          \item \textbf{High Storage Overhead:}\\
            Existing predictive policies require dedicated SRAM tables (2KB in SHiP) or complex sampling logic.
          \item \textbf{Lack of Zero-Overhead Resilience:}\\
            Production caches cannot afford dedicated SRAM tables or backward core-throttling buses.
          \item \textbf{Our Proposed Solution:}\\
            A \textbf{Zero-Overhead} policy (\texttt{PA-SRRIP}) and a \textbf{0.75-Byte Self-Tuning} controller (\texttt{PA-Adaptive}) that eliminates prefetch pollution right at insertion.
        \end{enumerate}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

        {\footnotesize\textbf{1. Demoted Initial Insertion (DII):}}
        \begin{itemize}\setlength{\itemsep}{0.05em}\scriptsize
          \item Hardware reads packet wire: \texttt{type == PREFETCH}.
          \item Demand fills $\rightarrow$ Insert at \textbf{$RRPV = 2$}.
          \item Prefetches $\rightarrow$ Demoted to \textbf{$RRPV = 3$}.
          \item \textbf{Why:} Inaccurate lines become immediate eviction victims on the very next miss!
        \end{itemize}
        
        \vspace{0.04cm}
        {\footnotesize\textbf{2. Hit-Triggered Promotion (HTP):}}
        \begin{itemize}\setlength{\itemsep}{0.05em}\scriptsize
          \item Useful prefetches hit before eviction.
          \item First demand hit promotes line to \textbf{$RRPV = 0$}.
          \item Shielded from eviction as verified demand data.
        \end{itemize}
        
        \vspace{0.04cm}
        {\footnotesize\textbf{3. Storage Overhead: STRICTLY ZERO BITS:}}
        \begin{itemize}\setlength{\itemsep}{0.05em}\scriptsize
          \item Uses existing 2-bit RRPV counters per line.
          \item No extra bits stored in cache tag directory!
        \end{itemize}
      \end{tcolorbox}

\begin{frame}{8. Architectural Design Extensions \& Exploration}
  \begin{columns}[t]
    % Column 1
    \begin{column}{0.49\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.2mm, bottom=1.2mm]
        {\small\textbf{\color{NavyPrimary}1. Bandwidth-Aware Throttling}}\par\vspace{0.04cm}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item \textbf{Concept:} Couple LLC insertion depth with DDR4 bus queue saturation monitors.
          \item \textbf{Mechanism:} When memory bus utilization exceeds 85\%, dynamically throttle L2 prefetch generation rate to save bandwidth.
        \end{itemize}
      \end{tcolorbox}

      \vspace{0.10cm}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.2mm, bottom=1.2mm]
        {\small\textbf{\color{NavyPrimary}2. Hybrid Way-Partitioning}}\par\vspace{0.04cm}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item \textbf{Concept:} Combine coarse-grained way-masking (Intel CAT) with line-level PA-SRRIP.
          \item \textbf{Mechanism:} Reserve 4 ways for latency-critical compute threads while applying PA-SRRIP across the remaining 12 shared ways.
        \end{itemize}
      \end{tcolorbox}
    \end{column}

    % Column 2
    \begin{column}{0.49\linewidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.2mm, bottom=1.2mm]
        {\small\textbf{\color{NavyPrimary}3. Dead-Block Signature Integration}}\par\vspace{0.04cm}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item \textbf{Concept:} Integrate lightweight PC hashing for demand lines while retaining zero-overhead prefetch demotion.
          \item \textbf{Benefit:} Identifies dead demand blocks without needing SHiP's full 2KB SRAM table.
        \end{itemize}
      \end{tcolorbox}

      \vspace{0.10cm}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=BoxBg, colframe=BoxBorder, top=1.2mm, bottom=1.2mm]
        {\small\textbf{\color{NavyPrimary}4. Cross-Core Prefetch Sharing}}\par\vspace{0.04cm}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item \textbf{Concept:} Enable collaborative multi-threaded prefetching.
          \item \textbf{Mechanism:} If Core 0 prefetches a line that Core 1 accesses, immediately promote to MRU ($R=0$) and mark as shared.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

        \begin{enumerate}\setlength{\itemsep}{0.2em}\footnotesize
          \item \textbf{Identified Bottleneck:}\\
            Proved that prefetch-oblivious LLC policies are the primary vector of cross-core cache thrashing in shared multicore systems.
          \item \textbf{Zero-Overhead Mechanism:}\\
            Designed \texttt{PA-SRRIP} using existing bus request wires and 2-bit RRPV counters, requiring \textbf{0 extra bits of SRAM storage}.
          \item \textbf{Adaptive Dynamic Controller:}\\
            Designed \texttt{PA-Adaptive} using a single 6-bit counter (\textbf{0.75 Bytes}) to handle dynamic phase-change behavior.
