\begin{frame}{Motivation \& Problem Statement}
  \begin{columns}[c]
    \begin{column}{0.47\linewidth}
      \begin{block}{1. Cache Thrashing by Prefetchers}
        \begin{itemize}
          \item Hardware prefetchers (AMPM, IP-Stride) flood the LLC on irregular pointer-chasing patterns (\texttt{429.mcf}).
          \item LRU and SRRIP treat speculative prefetches with high priority, evicting demand data.
          \item Causes severe LLC pollution and core IPC degradation under shared cache contention.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.47\linewidth}
      \begin{block}{2. Overhead of Existing Solutions}
        \begin{itemize}
          \item \textbf{DRRIP:} Requires PSEL set-dueling counters.
          \item \textbf{SHiP:} Requires an 8,192-entry SHCT predictor table.
          \item \textbf{High Hardware Cost:} Storage, area, and power overheads.
        \end{itemize}
      \end{block}

      \begin{block}{Research Goal}
        Achieve prefetch resilience with \textbf{ZERO hardware storage overhead} (0 extra metadata bits).
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------------------
% Slide 3: Proposed Policy
% ------------------------------------------------------------------------------
\begin{frame}{Proposed Prefetch-Aware LLC Policy}
  \begin{columns}[c]
    \begin{column}{0.47\linewidth}
      \begin{block}{Key Architectural Mechanism}
        \begin{enumerate}
          \item \textbf{Demoted Initial Insertion:}
            \begin{itemize}
              \item Demand Fills $\rightarrow$ MRU / $RRPV=2$.
              \item Prefetch Fills $\rightarrow$ LRU / $RRPV=3$.
            \end{itemize}
          \item \textbf{Self-Adaptive Promotion:}
            \begin{itemize}
              \item Useful prefetches hit before eviction $\rightarrow$ Promoted to MRU / $RRPV=0$.
              \item Useless prefetches fall off naturally.
            \end{itemize}
        \end{enumerate}
      \end{block}

      \begin{block}{Zero Hardware Overhead}
        Uses ChampSim's \texttt{access\_type::PREFETCH} hint ($0$ extra storage counters).
      \end{block}
    \end{column}
