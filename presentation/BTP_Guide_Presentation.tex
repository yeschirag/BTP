        {\footnotesize\textbf{\color{SuccessGreen}Streaming Workloads (\texttt{gcc}, Array Loops)}}\par\vspace{0.06cm}
        \begin{itemize}\setlength{\itemsep}{0.20em}\scriptsize
          \item \textbf{Access Pattern:} Predictable, sequential stride access.
          \item \textbf{Prefetch Accuracy:} \textbf{$>50\%$ Useful Hits}.
          \item \textbf{Latency Masking:} Successfully hides 200--250 cycle off-chip DRAM latency.
          \item \textbf{Hardware Impact:} Substantially accelerates core IPC.
        \end{itemize}
      \end{tcolorbox}
    \end{column}%
    \hfill%
    % Right Card: Irregular Workload
    \begin{column}{0.485\textwidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=CardBg, colframe=DangerRed!50, top=1.4mm, bottom=1.4mm, left=2.0mm, right=2.0mm]
        {\footnotesize\textbf{\color{DangerRed}Irregular Workloads (\texttt{mcf}, Graph Traversals)}}\par\vspace{0.06cm}
        \begin{itemize}\setlength{\itemsep}{0.20em}\scriptsize
          \item \textbf{Access Pattern:} Pointer-chasing, sparse graph/tree nodes.
          \item \textbf{Prefetch Accuracy:} \textbf{$<1\%$ Accuracy (Catastrophic Noise)}.
          \item \textbf{Cache Thrashing:} Speculative lines flood the cache set.
          \item \textbf{Hardware Impact:} Severe performance collapse.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}

  \vspace{0.20cm}

  % Bottom Alert Box: Multicore Thrashing
  \begin{tcolorbox}[boxrule=0.8pt, arc=1.8mm, colback=HighlightBg, colframe=HighlightBorder, top=1.5mm, bottom=1.5mm, left=2.5mm, right=2.5mm]
    {\footnotesize\textbf{\color{BrandBlue}The Multicore Shared LLC Interference Dilemma:}}\par\vspace{0.04cm}
    \scriptsize
    Under standard LRU replacement, all lines enter at the \textbf{MRU position}. When \texttt{mcf} generates noisy prefetches, they insert as MRU and immediately evict latency-critical demand data belonging to \texttt{gcc}.
    \textbf{Result:} Speculative noise destroys multicore throughput and fairness!
  \end{tcolorbox}
\end{frame}

        {\small\textbf{\color{NavyDark}Why PA-Adaptive is Robust}}\par\vspace{0.06cm}
        \begin{itemize}\setlength{\itemsep}{0.18em}\footnotesize
          \item \textbf{Total Overhead: Strictly 0.75 Bytes:}\\
            A single 6-bit counter for the entire 8MB LLC ($<30\,\mu\text{m}^2$ area).
          \item \textbf{Phase Change Resilience:}\\
            Dynamically adapts when workload alternates between linear vectors and pointer lists.
          \item \textbf{Asymmetric Reward/Penalty (+3 / -1):}\\
            Useful hits are rare and valuable ($+3$); fills are abundant ($-1$). Prevents false throttling.
          \item \textbf{Zero Critical-Path Latency:}\\
            Updates asynchronously when lines retire; zero delay on the 20-cycle tag read.
        \end{itemize}

\begin{frame}{8. Architectural Design Extensions \& Exploration}
  \vspace{-0.15cm}
  \begin{columns}[t]
    % Column 1
    \begin{column}{0.485\textwidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=CardBg, colframe=CardBorder, top=1.2mm, bottom=1.2mm]
        {\small\textbf{\color{NavyDark}1. Bandwidth-Aware Throttling}}\par\vspace{0.04cm}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item \textbf{Concept:} Couple LLC insertion depth with DDR4 bus queue saturation monitors.
          \item \textbf{Mechanism:} When memory bus utilization exceeds 85\%, dynamically throttle L2 prefetch generation rate to save bandwidth.
        \end{itemize}
      \end{tcolorbox}

      \vspace{0.10cm}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=CardBg, colframe=CardBorder, top=1.2mm, bottom=1.2mm]
        {\small\textbf{\color{NavyDark}2. Hybrid Way-Partitioning}}\par\vspace{0.04cm}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item \textbf{Concept:} Combine coarse-grained way-masking (Intel CAT) with line-level PA-SRRIP.
          \item \textbf{Mechanism:} Reserve 4 ways for latency-critical compute threads while applying PA-SRRIP across the remaining 12 shared ways.
        \end{itemize}
      \end{tcolorbox}
    \end{column}%
    \hfill%
    % Column 2
    \begin{column}{0.485\textwidth}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=CardBg, colframe=CardBorder, top=1.2mm, bottom=1.2mm]
        {\small\textbf{\color{NavyDark}3. Dead-Block Signature Integration}}\par\vspace{0.04cm}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item \textbf{Concept:} Integrate lightweight PC hashing for demand lines while retaining zero-overhead prefetch demotion.
          \item \textbf{Benefit:} Identifies dead demand blocks without needing SHiP's full 2KB SRAM table.
        \end{itemize}
      \end{tcolorbox}

      \vspace{0.10cm}
      \begin{tcolorbox}[boxrule=0.7pt, arc=1.5mm, colback=CardBg, colframe=CardBorder, top=1.2mm, bottom=1.2mm]
        {\small\textbf{\color{NavyDark}4. Cross-Core Prefetch Sharing}}\par\vspace{0.04cm}
        \begin{itemize}\setlength{\itemsep}{0.12em}\scriptsize
          \item \textbf{Concept:} Enable collaborative multi-threaded prefetching.
          \item \textbf{Mechanism:} If Core 0 prefetches a line that Core 1 accesses, immediately promote to MRU ($R=0$) and mark as shared.
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}
